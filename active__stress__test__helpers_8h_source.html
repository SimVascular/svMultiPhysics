51/**
52 * @brief Run a twitch experiment on the ActiveStress model passed as template
53 * argument and compare the simulated trajectory with a trusted external
54 * reference solution loaded from a CSV file.
55 *
56 * The twitch experiment uses a prescribed calcium transient to trigger
57 * contraction in the active stress model (see @ref active_stress_calcium_transient).
58 * Fiber stretch is also prescribed (see @ref active_stress_fiber_stretch), and
59 * fiber stretch rate is obtained from the prescribed stretch trajectory.
60 *
61 * The active stress model is solved standalone, i.e. only one system of ODEs
62 * is solved rather than a coupled three-dimensional mechanics problem.
63 * Therefore, the test verifies the active stress ODE definition and time
64 * stepping independently of the mechanics solver.
65 *
66 * @anchor active_stress_calcium_transient
67 * ### Prescribed calcium transient
68 *
69 * The calcium concentration is
70 * @f[
71 * C(t) =
72 * \begin{cases}
73 * C_0, & t < t_0, \\[4pt]
74 * C_0 + (C_{\max}-C_0)
75 * \dfrac{
76 * e^{-(t-t_0)/\tau_d} - e^{-(t-t_0)/\tau_r}
77 * }{
78 * e^{-t_p/\tau_d} - e^{-t_p/\tau_r}
79 * }, & t \ge t_0,
80 * \end{cases}
81 * @f]
82 * where
83 * @f[
84 * C_0 = 10^{-4}\,\mathrm{mM}, \quad
85 * C_{\max} = 9\times10^{-4}\,\mathrm{mM}, \quad
86 * \tau_r = 20\,\mathrm{ms}, \quad
87 * \tau_d = 50\,\mathrm{ms}, \quad
88 * t_0 = 10\,\mathrm{ms},
89 * @f]
90 * and
91 * @f[
92 * t_p =
93 * \frac{\log(\tau_r/\tau_d)\tau_r\tau_d}
94 * {\tau_r-\tau_d}.
95 * @f]
96 *
97 * @anchor active_stress_fiber_stretch
98 * ### Prescribed fiber stretch
99 *
100 * Fiber stretch is defined as
101 * @f[
102 * \lambda(t) = \frac{L(t)}{L_0},
103 * @f]
104 * with
105 * @f[
106 * L(t) =
107 * \begin{cases}
108 * L_0, & t < t_s, \\[4pt]
109 * L_0 - \dfrac{L_0-L_{\min}}{2}
110 * \left[
111 * 1-\cos\left(\pi\frac{t-t_s}{t_m-t_s}\right)
112 * \right],
113 * & t_s \le t \le t_m, \\[8pt]
114 * L_{\min} + \dfrac{L_0-L_{\min}}{2}
115 * \left[
116 * 1-\cos\left(\pi\frac{t-t_m}{t_r-t_m}\right)
117 * \right],
118 * & t_m < t \le t_r, \\[8pt]
119 * L_0, & t > t_r,
120 * \end{cases}
121 * @f]
122 * where
123 * @f[
124 * L_0 = 2.2\,\mu\mathrm{m}, \quad
125 * L_{\min} = 2.134\,\mu\mathrm{m}, \quad
126 * t_s = 30\,\mathrm{ms}, \quad
127 * t_m = 150\,\mathrm{ms}, \quad
128 * t_r = 350\,\mathrm{ms}.
129 * @f]
130 *
131 * ### Format of the reference solution
132 *
133 * The reference CSV contains a header followed by reference values at selected
134 * simulation steps. The first two columns are @c step and @c Ta, containing the
135 * simulation step and active tension, respectively. If the model has additional
136 * internal state variables, their values follow in subsequent columns.
137 *
138 * The state variables are assumed by default to have the same order as in the
139 * concrete model being tested. If necessary, @c state_reference_columns
140 * supplies the CSV column name corresponding to each state variable, in the
141 * concrete model's state order.
142 *
143 * Comparisons are performed only at the checkpoints listed in the reference
144 * CSV; the file does not need to contain every simulated time step. At each
145 * checkpoint, active tension and each state variable are compared pointwise
146 * with the corresponding reference value. If a pointwise difference exceeds
147 * the prescribed tolerance, GoogleTest records a failed expectation and the
148 * test fails.
149 *
150 * Checkpoints are integer simulation-step indices and must coincide with
151 * simulated time steps; intermediate checkpoints are not interpolated.
152 * Checkpoint @c N is compared after the update starting at
153 * @f$ t = N \Delta t @f$, which advances the interval
154 * @f$[N\Delta t,(N+1)\Delta t]@f$.
155 */
156template <class ConcreteModel>
157class ActiveStressTrajectoryTest {
158 static_assert(std::is_base_of_v<ActiveStress, ConcreteModel>,
159 "ConcreteModel must derive from ActiveStress");

162 /// operations to the trajectory test.

166 using ConcreteModel::compute_active_tension_local;
167 using ConcreteModel::init_local;
168 using ConcreteModel::read_model_specific_parameters;
169 };

179 /**
180 * @brief Configure a standalone trajectory test and load its reference CSV.
181 *
182 * @param parameters Model-specific parameters.
183 * @param configuration Trajectory inputs and reference-data mapping.
184 */
185 ActiveStressTrajectoryTest(
186 const Parameters &parameters,
187 ActiveStressTrajectoryConfiguration configuration)
188 : model_(make_configured_model(parameters)),
189 configuration_(std::move(configuration)),
190 n_steps_(number_of_steps(configuration_.final_time,
191 configuration_.time_step))
192 {
193 validate_configuration();
194 reference_path_ = std::string(UNIT_TEST_DATA_DIR) + "/" +
195 configuration_.reference_csv_filename;
196 checkpoints_ = load_reference_data(
197 reference_path_, model_->n_states, n_steps_,
198 configuration_.state_reference_columns);
199 }

201 /** @brief Execute the common twitch protocol and compare every checkpoint. */
202 void run()
203 {
204 Vector<double> state(model_->n_states);
205 model_->init_local(state);
206
207 size_t checkpoint_index = 0;
208 for (int step = 0; step < n_steps_; ++step) {
209 const double time = step * configuration_.time_step;
210 const double calcium = calcium_at(time);
211 const double fiber_stretch = fiber_stretch_at(time);
212 const double fiber_stretch_rate =
213 (fiber_stretch_at(time + configuration_.time_step) - fiber_stretch) /
214 configuration_.time_step;
215
216 model_->advance_time_step_local(time, configuration_.time_step, calcium,
217 fiber_stretch, fiber_stretch_rate, state);
218 const double active_tension =
219 model_->compute_active_tension_local(state, fiber_stretch);
220
221 compare_checkpoint_if_present(step, state, active_tension,
222 checkpoint_index);
223 }
224
225 EXPECT_EQ(checkpoint_index, checkpoints_.size())
226 << "Not all reference checkpoints were visited in " << reference_path_;
227 }

251 /// Validate the trajectory inputs and resolve default state-column names.
252 void validate_configuration()
253 {
254 svmp::throw_if<svmp::FE::InvalidArgumentException>(
255 !(configuration_.tolerance > 0.0) ||
256 !std::isfinite(configuration_.tolerance),
257 "ActiveStress test tolerance must be finite and positive");
258 svmp::throw_if<svmp::FE::InvalidArgumentException>(
259 configuration_.reference_csv_filename.empty(),
260 "ActiveStress trajectory reference CSV filename must not be empty");
261
262 if (configuration_.state_reference_columns.empty()) {
263 configuration_.state_reference_columns.reserve(model_->n_states);
264 for (unsigned int state = 0; state < model_->n_states; ++state)
265 configuration_.state_reference_columns.push_back(
266 "s" + std::to_string(state));
267 }
268
269 svmp::throw_if<svmp::FE::InvalidArgumentException>(
270 configuration_.state_reference_columns.size() != model_->n_states,
271 "ActiveStress reference state-column mapping has the wrong size");
272
273 for (size_t i = 0; i < configuration_.state_reference_columns.size(); ++i) {
274 const auto &name = configuration_.state_reference_columns[i];
275 svmp::throw_if<svmp::FE::InvalidArgumentException>(
276 name.empty(),
277 "ActiveStress reference state-column names must not be empty");
278 svmp::throw_if<svmp::FE::InvalidArgumentException>(
279 name == "step",
280 "ActiveStress reference state-column mapping cannot use step");
281 svmp::throw_if<svmp::FE::InvalidArgumentException>(
282 std::find(configuration_.state_reference_columns.begin(),
283 configuration_.state_reference_columns.begin() + i,
284 name) !=
285 configuration_.state_reference_columns.begin() + i,
286 "ActiveStress reference state-column names must be unique");
287 }
288 }

290 /// Validate the time configuration and return the number of simulation steps.
291 static int number_of_steps(double final_time, double time_step)
292 {
293 svmp::throw_if<svmp::FE::InvalidArgumentException>(
294 !(final_time > 0.0) || !std::isfinite(final_time),
295 "ActiveStress test final time must be finite and positive");
296 svmp::throw_if<svmp::FE::InvalidArgumentException>(
297 !(time_step > 0.0) || !std::isfinite(time_step),
298 "ActiveStress test time step must be finite and positive");
299
300 const double step_count = final_time / time_step;
301 const double rounded_step_count = std::round(step_count);
302 const double scale = std::max(1.0, std::fabs(step_count));
303 svmp::throw_if<svmp::FE::InvalidArgumentException>(
304 std::fabs(step_count - rounded_step_count) > 1.0e-12 * scale,
305 "ActiveStress test final time must be an integer multiple of the time step");
306 svmp::throw_if<svmp::FE::InvalidArgumentException>(
307 rounded_step_count > std::numeric_limits<int>::max(),
308 "ActiveStress test has too many time steps");
309
310 return static_cast<int>(rounded_step_count);
311 }

313 /// Remove leading and trailing whitespace.
314 static std::string trim(const std::string &value)
315 {
316 size_t begin = 0;
317 while (begin < value.size() &&
318 std::isspace(static_cast<unsigned char>(value[begin])))
319 ++begin;
320
321 size_t end = value.size();
322 while (end > begin &&
323 std::isspace(static_cast<unsigned char>(value[end - 1])))
324 --end;
325
326 return value.substr(begin, end - begin);
327 }

329 /// Split a CSV row, trim its fields, and preserve a trailing empty field.
330 static std::vector<std::string> split_csv_row(const std::string &line)
331 {
332 std::vector<std::string> fields;
333 std::istringstream stream(line);
334 std::string field;
335 while (std::getline(stream, field, ','))
336 fields.push_back(trim(field));
337 if (!line.empty() && line.back() == ',')
338 fields.emplace_back();
339 return fields;
340 }

342 /// Parse one finite floating-point value from a CSV field.
343 static double parse_double(const std::string &field,
344 const std::string &path,
345 size_t line_number)
346 {
347 size_t parsed_characters = 0;
348 double value = 0.0;
349 try {
350 value = std::stod(field, &parsed_characters);
351 } catch (const std::exception &) {
352 svmp::raise<svmp::ParseException>(
353 "Invalid number in " + path + " at line " +
354 std::to_string(line_number));
355 }
356
357 svmp::throw_if<svmp::ParseException>(
358 parsed_characters != field.size() || !std::isfinite(value),
359 "Invalid number in " + path + " at line " +
360 std::to_string(line_number));
361 return value;
362 }

364 /// Load and validate the reference checkpoints and state-column mapping.
365 static std::vector<Checkpoint>
366 load_reference_data(const std::string &path,
367 unsigned int n_states,
368 int n_steps,
369 const std::vector<std::string> &requested_state_columns)
370 {
371 std::ifstream file(path);
372 svmp::throw_if<svmp::FileNotFoundException>(!file.is_open(), path);
373
374 std::vector<std::string> header;
375 std::vector<size_t> state_columns;
376 std::vector<Checkpoint> checkpoints;
377 std::string line;
378 size_t line_number = 0;
379
380 while (std::getline(file, line)) {
381 ++line_number;
382 const std::string stripped_line = trim(line);
383 if (stripped_line.empty() || stripped_line[0] == '#')
384 continue;
385
386 if (header.empty()) {
387 header = split_csv_row(stripped_line);
388 svmp::throw_if<svmp::ParseException>(
389 header.size() < 2 || header[0] != "step" || header[1] != "Ta",
390 "ActiveStress reference CSV must begin with columns step,Ta: " +
391 path);
392
393 for (size_t i = 0; i < header.size(); ++i) {
394 svmp::throw_if<svmp::ParseException>(
395 header[i].empty(),
396 "ActiveStress reference CSV has an empty column name: " + path);
397 svmp::throw_if<svmp::ParseException>(
398 std::find(header.begin(), header.begin() + i, header[i]) !=
399 header.begin() + i,
400 "ActiveStress reference CSV has duplicate column " + header[i] +
401 ": " + path);
402 }
403
404 state_columns.reserve(n_states);
405 for (const std::string &column_name : requested_state_columns) {
406 const auto column = std::find(header.begin(), header.end(), column_name);
407 svmp::throw_if<svmp::ParseException>(
408 column == header.end(),
409 "ActiveStress reference CSV is missing state column " +
410 column_name + ": " + path);
411 state_columns.push_back(
412 static_cast<size_t>(std::distance(header.begin(), column)));
413 }
414 continue;
415 }
416
417 const auto fields = split_csv_row(stripped_line);
418 svmp::throw_if<svmp::ParseException>(
419 fields.size() != header.size(),
420 "Wrong column count in " + path + " at line " +
421 std::to_string(line_number));
422
423 const double step_value =
424 parse_double(fields[0], path, line_number);
425 const double rounded_step = std::round(step_value);
426 svmp::throw_if<svmp::ParseException>(
427 std::fabs(step_value - rounded_step) > 1.0e-12 ||
428 rounded_step < 0.0 || rounded_step >= n_steps,
429 "Invalid checkpoint step in " + path + " at line " +
430 std::to_string(line_number));
431
432 Checkpoint checkpoint;
433 checkpoint.step = static_cast<int>(rounded_step);
434 checkpoint.active_tension =
435 parse_double(fields[1], path, line_number);
436 checkpoint.state.reserve(n_states);
437 for (size_t column : state_columns)
438 checkpoint.state.push_back(
439 parse_double(fields[column], path, line_number));
440
441 svmp::throw_if<svmp::ParseException>(
442 !checkpoints.empty() &&
443 checkpoint.step <= checkpoints.back().step,
444 "ActiveStress reference checkpoints must be strictly increasing: " +
445 path);
446 checkpoints.push_back(std::move(checkpoint));
447 }
448
449 svmp::throw_if<svmp::ParseException>(
450 header.empty(), "ActiveStress reference CSV has no header: " + path);
451 svmp::throw_if<svmp::ParseException>(
452 checkpoints.empty(),
453 "ActiveStress reference CSV has no checkpoints: " + path);
454 return checkpoints;
455 }

457 /// Evaluate the prescribed calcium transient at @p time.
458 static double calcium_at(double time)
459 {
460 constexpr double baseline_calcium = 1.0e-4;
461 constexpr double peak_calcium = 9.0e-4;
462 constexpr double rise_time = 20.0;
463 constexpr double decay_time = 50.0;
464 constexpr double onset_time = 10.0;
465
466 if (time < onset_time)
467 return baseline_calcium;
468
469 const double peak_time =
470 std::log(rise_time / decay_time) * rise_time * decay_time /
471 (rise_time - decay_time);
472 const double peak_raw_factor =
473 std::exp(-peak_time / decay_time) -
474 std::exp(-peak_time / rise_time);
475 const double elapsed = time - onset_time;
476 const double raw_calcium =
477 std::exp(-elapsed / decay_time) - std::exp(-elapsed / rise_time);
478 return baseline_calcium +
479 (peak_calcium - baseline_calcium) * raw_calcium / peak_raw_factor;
480 }
481
482 /// Evaluate the prescribed fiber stretch at @p time.
483 double fiber_stretch_at(double time) const
484 {
485 constexpr double resting_sarcomere_length = 2.2;
486 constexpr double minimum_sarcomere_length = 2.134;
487 constexpr double shortening_onset = 30.0;
488 constexpr double minimum_time = 150.0;
489 constexpr double recovery_time = 350.0;
490
491 double sarcomere_length = resting_sarcomere_length;
492 if (time >= shortening_onset && time <= minimum_time) {
493 const double fraction =
494 (time - shortening_onset) / (minimum_time - shortening_onset);
495 sarcomere_length =
496 resting_sarcomere_length -
497 (resting_sarcomere_length - minimum_sarcomere_length) *
498 (1.0 - std::cos(std::numbers::pi * fraction)) / 2.0;
499 } else if (time > minimum_time && time <= recovery_time) {
500 const double fraction =
501 (time - minimum_time) / (recovery_time - minimum_time);
502 sarcomere_length =
503 minimum_sarcomere_length +
504 (resting_sarcomere_length - minimum_sarcomere_length) *
505 (1.0 - std::cos(std::numbers::pi * fraction)) / 2.0;
506 }
507
508 return sarcomere_length / resting_sarcomere_length;
509 }

511 /// Return the pointwise tolerance @f$\mathrm{tol}(1 + |expected|)@f$.
512 double scaled_tolerance(double expected) const
513 {
514 return configuration_.tolerance * (1.0 + std::fabs(expected));
515 }

517 /// Compare the checkpoint matching @p step, if one is listed.
518 void compare_checkpoint_if_present(int step,
519 const Vector<double> &state,
520 double active_tension,
521 size_t &checkpoint_index) const
522 {
523 if (checkpoint_index >= checkpoints_.size() ||
524 checkpoints_[checkpoint_index].step != step)
525 return;
526
527 compare_checkpoint(checkpoints_[checkpoint_index], state, active_tension);
528 ++checkpoint_index;
529 }

531 /// Compare active tension and every state value at one checkpoint.
532 void compare_checkpoint(const Checkpoint &checkpoint,
533 const Vector<double> &state,
534 double active_tension) const
535 {
536 EXPECT_NEAR(active_tension, checkpoint.active_tension,
537 scaled_tolerance(checkpoint.active_tension))
538 << "checkpoint at trajectory update index " << checkpoint.step
539 << ", active tension column Ta";
540
541 ASSERT_EQ(state.size(), checkpoint.state.size())
542 << "checkpoint at trajectory update index " << checkpoint.step
543 << ", state-vector size";
544 for (unsigned int index = 0; index < state.size(); ++index) {
545 EXPECT_NEAR(state[index], checkpoint.state[index],
546 scaled_tolerance(checkpoint.state[index]))
547 << "checkpoint at trajectory update index " << checkpoint.step
548 << ", state[" << index << "] column "
549 << configuration_.state_reference_columns[index];
550 }
551 }
