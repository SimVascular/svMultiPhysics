29/**
30 * @brief Configure one standalone trusted-reference IonicModel trajectory.
31 *
32 * The configuration selects the public integration inputs and maps every
33 * model state to a column in the external reference CSV.
34 */
35struct IonicModelTrajectoryConfiguration {
36 /// Integration method passed to IonicModel::integ; must be FE, RK4, or CN2.
37 TimeIntegrationType integration_type = TimeIntegrationType::NA;
38
39 /// Positive zone identifier passed to every trajectory update.
40 int zone_id = 1;
41
42 /// Time step passed to IonicModel::integ and used to compute update times.
43 double time_step = 0.0;
44
45 /// Number of calls to IonicModel::integ.
46 std::size_t update_count = 0;
47
48 /// Optional complete X replacement applied after IonicModel::init.
49 std::optional<std::vector<double>> initial_X_override;
50
51 /// Optional complete Xg replacement applied after IonicModel::init.
52 std::optional<std::vector<double>> initial_Xg_override;
53
54 /// Reference filename under UNIT_TEST_DATA_DIR.
55 std::string reference_csv_filename;
56
57 /**
58 * @brief CSV column name for every model state.
59 *
60 * Entries correspond in order to @c X[0] through @c X[nX-1], followed by
61 * @c Xg[0] through @c Xg[nG-1]. Each entry names the CSV column containing
62 * the reference value for that state; the column name need not match the
63 * model state name. Every model state requires an entry, but the physical
64 * CSV column order may differ from this mapping order.
65 */
66 std::vector<std::string> state_reference_columns;
67
68 /// Base tolerance in the pointwise comparison formula.
69 double tolerance = 1.0e-10;
70
71 /// Constant stimulus used when stimulus_at_time is not set.
72 double stimulus = 0.0;
73
74 /**
75 * @brief Optional stimulus evaluated once per update at the old-state time.
76 *
77 * When set, this function supplies the stimulus instead of the constant
78 * value above.
79 */
80 std::function<double(double)> stimulus_at_time;
81
82 /// Constant SAC coefficient passed to every trajectory update.
83 double sac_coefficient = 0.0;
84};

86/**
87 * @brief Run a standalone IonicModel trajectory and compare it with a trusted
88 * external reference solution loaded from a CSV file.
89 *
90 * The concrete model type is given by @c ConcreteModel and is used only for
91 * construction and parameter typing. The test allocates @c X and @c Xg, calls
92 * IonicModel::init, applies any complete state overrides, and advances the
93 * model through the public IonicModel::integ interface for the configured
94 * number of updates. Only the ionic model is advanced; no coupled tissue or
95 * mechanics problem is solved. Reference values must use the representation
96 * exposed by the public interface; this helper intentionally performs no
97 * voltage, time, or state transformations.
98 *
99 * If @c stimulus_at_time is set, it is evaluated once at the left endpoint of
100 * each update and its result is used for the complete call to
101 * IonicModel::integ. Otherwise, the constant @c stimulus value is used.
102 *
103 * ### Format of the reference solution
104 *
105 * The reference CSV contains a header followed by values at selected
106 * trajectory checkpoints. The first column is @c step. Entries in
107 * @c state_reference_columns map in order to @c X[0] through @c X[nX-1], then
108 * @c Xg[0] through @c Xg[nG-1]; each entry names the CSV column containing the
109 * corresponding state value. Those columns may occur in any physical order
110 * after @c step; additional columns are ignored.
111 *
112 * Comparisons are performed only at the checkpoints listed in the CSV; the
113 * file does not need to contain every trajectory update. Checkpoints must be
114 * nonnegative integer update counts, strictly increasing, and less than or
115 * equal to @c update_count. Intermediate checkpoints are not interpolated.
116 *
117 * Checkpoint N is the state after exactly N completed updates. In particular,
118 * checkpoint 0 is compared immediately after init() and optional state
119 * overrides, before the first call to integ(). Each update starting at N uses
120 * t = N * time_step, so the resulting state is checkpoint N + 1.
121 *
122 * At each checkpoint, every @c X and @c Xg value is compared pointwise using
123 * the tolerance @f$\mathrm{tol}(1+|expected|)@f$. A difference exceeding that
124 * tolerance records a failed GoogleTest expectation.
125 */
126template <class ConcreteModel>
127class IonicModelTrajectoryTest {
128 static_assert(std::is_base_of_v<IonicModel, ConcreteModel>,
129 "ConcreteModel must derive from IonicModel");
130
131public:
132 /// Model-specific parameter type used to construct the concrete model.
133 using Parameters = typename ConcreteModel::Parameters;
134
135 static_assert(std::is_base_of_v<IonicModelParameters, Parameters>,
136 "ConcreteModel::Parameters must derive from "
137 "IonicModelParameters");

139 /**
140 * @brief Configure the model, validate the trajectory inputs, and load the
141 * reference CSV.
142 *
143 * @param parameters Parameters for the concrete ionic model.
144 * @param configuration Trajectory inputs and reference-data mapping.
145 */
146 IonicModelTrajectoryTest(
147 const Parameters &parameters,
148 IonicModelTrajectoryConfiguration configuration)
149 : model_(make_configured_model(parameters)),
150 configuration_(std::move(configuration))
151 {
152 validate_configuration();
153 reference_path_ = std::string(UNIT_TEST_DATA_DIR) + "/" +
154 configuration_.reference_csv_filename;
155 checkpoints_ = load_reference_data();
156 }
157
158 /** @brief Initialize, integrate, and compare every reference checkpoint. */
159 void run()
160 {
161 Vector<double> X(model_->nX());
162 Vector<double> Xg(model_->nG());
163 model_->init(X, Xg);
164
165 apply_override(configuration_.initial_X_override, X);
166 apply_override(configuration_.initial_Xg_override, Xg);
167
168 odeType solver;
169 solver.tIntType = configuration_.integration_type;
170
171 std::size_t checkpoint_index = 0;
172 compare_checkpoint_if_present(/* completed_updates = */ 0, X, Xg,
173 checkpoint_index);
174
175 for (std::size_t completed_updates = 1;
176 completed_updates <= configuration_.update_count;
177 ++completed_updates) {
178 const double time =
179 static_cast<double>(completed_updates - 1) *
180 configuration_.time_step;
181 const double stimulus = evaluate_stimulus(time);
182 model_->integ(solver, configuration_.zone_id, time,
183 configuration_.time_step, stimulus,
184 configuration_.sac_coefficient, X, Xg);
185 compare_checkpoint_if_present(completed_updates, X, Xg,
186 checkpoint_index);
187 }
188
189 EXPECT_EQ(checkpoint_index, checkpoints_.size())
190 << "Not all reference checkpoints were visited in " << reference_path_;
191 }

194 /// Reference state after a specified number of completed updates.
195 struct Checkpoint {
196 /// Number of completed trajectory updates.
197 std::size_t completed_updates;
198
199 /// Expected values ordered as all X entries followed by all Xg entries.
200 std::vector<double> state;
201 };

212 /// Validate all trajectory inputs and state-column mappings.
213 void validate_configuration() const
214 {
215 switch (configuration_.integration_type) {
216 case TimeIntegrationType::FE:
217 case TimeIntegrationType::RK4:
218 case TimeIntegrationType::CN2:
219 break;
220 default:
221 svmp::raise<svmp::FE::InvalidArgumentException>(
222 "IonicModel trajectory integration type must be FE, RK4, or CN2");
223 }
224
225 svmp::throw_if<svmp::FE::InvalidArgumentException>(
226 configuration_.zone_id <= 0,
227 "IonicModel trajectory zone_id must be positive");
228 svmp::throw_if<svmp::FE::InvalidArgumentException>(
229 !(configuration_.time_step > 0.0) ||
230 !std::isfinite(configuration_.time_step),
231 "IonicModel trajectory time step must be finite and positive");
232 svmp::throw_if<svmp::FE::InvalidArgumentException>(
233 configuration_.update_count == 0,
234 "IonicModel trajectory update count must be positive");
235 svmp::throw_if<svmp::FE::InvalidArgumentException>(
236 !std::isfinite(configuration_.time_step *
237 static_cast<double>(configuration_.update_count)),
238 "IonicModel trajectory final time must be finite");
239 svmp::throw_if<svmp::FE::InvalidArgumentException>(
240 !(configuration_.tolerance > 0.0) ||
241 !std::isfinite(configuration_.tolerance),
242 "IonicModel trajectory tolerance must be finite and positive");
243 svmp::throw_if<svmp::FE::InvalidArgumentException>(
244 !std::isfinite(configuration_.stimulus) ||
245 !std::isfinite(configuration_.sac_coefficient),
246 "IonicModel trajectory stimulus and SAC coefficient must be finite");
247 svmp::throw_if<svmp::FE::InvalidArgumentException>(
248 configuration_.reference_csv_filename.empty(),
249 "IonicModel trajectory reference CSV filename must not be empty");
250
251 validate_override(configuration_.initial_X_override, model_->nX(), "X");
252 validate_override(configuration_.initial_Xg_override, model_->nG(), "Xg");
253
254 const std::size_t state_count = model_->nX() + model_->nG();
255 svmp::throw_if<svmp::FE::InvalidArgumentException>(
256 configuration_.state_reference_columns.size() != state_count,
257 "IonicModel reference state-column mapping must contain all X "
258 "columns followed by all Xg columns");
259
260 for (std::size_t i = 0;
261 i < configuration_.state_reference_columns.size(); ++i) {
262 const auto &name = configuration_.state_reference_columns[i];
263 svmp::throw_if<svmp::FE::InvalidArgumentException>(
264 name.empty(),
265 "IonicModel reference state-column names must not be empty");
266 svmp::throw_if<svmp::FE::InvalidArgumentException>(
267 name == "step",
268 "IonicModel reference state-column mapping cannot use step");
269 svmp::throw_if<svmp::FE::InvalidArgumentException>(
270 std::find(configuration_.state_reference_columns.begin(),
271 configuration_.state_reference_columns.begin() + i,
272 name) !=
273 configuration_.state_reference_columns.begin() + i,
274 "IonicModel reference state-column names must be unique");
275 }
276 }

278 /// Validate the size and finiteness of an optional complete state override.
279 static void
280 validate_override(const std::optional<std::vector<double>> &values,
281 std::size_t expected_size,
282 const std::string &state_name)
283 {
284 if (!values)
285 return;
286 svmp::throw_if<svmp::FE::InvalidArgumentException>(
287 values->size() != expected_size,
288 "IonicModel initial " + state_name + " override has the wrong size");
289 svmp::throw_if<svmp::FE::InvalidArgumentException>(
290 !std::all_of(values->begin(), values->end(),
291 [](double value) { return std::isfinite(value); }),
292 "IonicModel initial " + state_name +
293 " override must contain finite values");
294 }

296 /// Evaluate the callback or constant stimulus and require a finite result.
297 double evaluate_stimulus(double time) const
298 {
299 const double stimulus = configuration_.stimulus_at_time
300 ? configuration_.stimulus_at_time(time)
301 : configuration_.stimulus;
302 svmp::throw_if<svmp::FE::InvalidArgumentException>(
303 !std::isfinite(stimulus),
304 "IonicModel trajectory stimulus must be finite at time " +
305 std::to_string(time));
306 return stimulus;
307 }

309 /// Apply an optional complete replacement of a model state vector.
310 static void
311 apply_override(const std::optional<std::vector<double>> &values,
312 Vector<double> &state)
313 {
314 if (!values)
315 return;
316 for (std::size_t index = 0; index < values->size(); ++index)
317 state[index] = (*values)[index];

320 /// Remove leading and trailing whitespace.
321 static std::string trim(const std::string &value)
322 {
323 std::size_t begin = 0;
324 while (begin < value.size() &&
325 std::isspace(static_cast<unsigned char>(value[begin])))
326 ++begin;
327
328 std::size_t end = value.size();
329 while (end > begin &&
330 std::isspace(static_cast<unsigned char>(value[end - 1])))
331 --end;
332
333 return value.substr(begin, end - begin);
334 }

336 /// Split a CSV row, trim its fields, and preserve a trailing empty field.
337 static std::vector<std::string> split_csv_row(const std::string &line)
338 {
339 std::vector<std::string> fields;
340 std::istringstream stream(line);
341 std::string field;
342 while (std::getline(stream, field, ','))
343 fields.push_back(trim(field));
344 if (!line.empty() && line.back() == ',')
345 fields.emplace_back();
346 return fields;
347 }

349 /// Parse one complete finite floating-point value from a CSV field.
350 static double parse_double(const std::string &field,
351 const std::string &path,
352 std::size_t line_number)
353 {
354 std::size_t parsed_characters = 0;
355 double value = 0.0;
356 try {
357 value = std::stod(field, &parsed_characters);
358 } catch (const std::exception &) {
359 svmp::raise<svmp::ParseException>(
360 "Invalid number in " + path + " at line " +
361 std::to_string(line_number));
362 }
363
364 svmp::throw_if<svmp::ParseException>(
365 parsed_characters != field.size() || !std::isfinite(value),
366 "Invalid number in " + path + " at line " +
367 std::to_string(line_number));
368 return value;
369 }

371 /// Load and validate the reference checkpoints and selected state values.
372 std::vector<Checkpoint> load_reference_data() const
373 {
374 std::ifstream file(reference_path_);
375 svmp::throw_if<svmp::FileNotFoundException>(!file.is_open(),
376 reference_path_);
377
378 std::vector<std::string> header;
379 std::vector<std::size_t> state_columns;
380 std::vector<Checkpoint> checkpoints;
381 std::string line;
382 std::size_t line_number = 0;
383
384 while (std::getline(file, line)) {
385 ++line_number;
386 const std::string stripped_line = trim(line);
387 if (stripped_line.empty() || stripped_line[0] == '#')
388 continue;
389
390 if (header.empty()) {
391 header = split_csv_row(stripped_line);
392 validate_header(header, state_columns);
393 continue;
394 }
395
396 const auto fields = split_csv_row(stripped_line);
397 svmp::throw_if<svmp::ParseException>(
398 fields.size() != header.size(),
399 "Wrong column count in " + reference_path_ + " at line " +
400 std::to_string(line_number));
401
402 const double completed_updates_value =
403 parse_double(fields[0], reference_path_, line_number);
404 const double rounded_completed_updates =
405 std::round(completed_updates_value);
406 svmp::throw_if<svmp::ParseException>(
407 std::fabs(completed_updates_value - rounded_completed_updates) >
408 1.0e-12 ||
409 rounded_completed_updates < 0.0,
410 "Invalid checkpoint step in " + reference_path_ + " at line " +
411 std::to_string(line_number));
412
413 svmp::throw_if<svmp::ParseException>(
414 rounded_completed_updates >
415 static_cast<double>(configuration_.update_count),
416 "Checkpoint step is outside the configured update range in " +
417 reference_path_ + " at line " + std::to_string(line_number));
418
419 Checkpoint checkpoint;
420 checkpoint.completed_updates =
421 static_cast<std::size_t>(rounded_completed_updates);
422 checkpoint.state.reserve(state_columns.size());
423 for (std::size_t column : state_columns)
424 checkpoint.state.push_back(
425 parse_double(fields[column], reference_path_, line_number));
426
427 svmp::throw_if<svmp::ParseException>(
428 !checkpoints.empty() &&
429 checkpoint.completed_updates <=
430 checkpoints.back().completed_updates,
431 "IonicModel reference checkpoints must be strictly increasing: " +
432 reference_path_);
433 checkpoints.push_back(std::move(checkpoint));
434 }
435
436 svmp::throw_if<svmp::ParseException>(
437 header.empty(),
438 "IonicModel reference CSV has no header: " + reference_path_);
439 svmp::throw_if<svmp::ParseException>(
440 checkpoints.empty(),
441 "IonicModel reference CSV has no checkpoints: " + reference_path_);
442 return checkpoints;
443 }

445 /// Validate the CSV header and resolve the configured state-column indices.
446 void validate_header(const std::vector<std::string> &header,
447 std::vector<std::size_t> &state_columns) const
448 {
449 svmp::throw_if<svmp::ParseException>(
450 header.size() < 2 || header.front() != "step",
451 "IonicModel reference CSV must begin with a step column: " +
452 reference_path_);
453
454 for (std::size_t i = 0; i < header.size(); ++i) {
455 svmp::throw_if<svmp::ParseException>(
456 header[i].empty(),
457 "IonicModel reference CSV has an empty column name: " +
458 reference_path_);
459 svmp::throw_if<svmp::ParseException>(
460 std::find(header.begin(), header.begin() + i, header[i]) !=
461 header.begin() + i,
462 "IonicModel reference CSV has duplicate column " + header[i] +
463 ": " + reference_path_);
464 }
465
466 state_columns.reserve(configuration_.state_reference_columns.size());
467 for (const std::string &column_name :
468 configuration_.state_reference_columns) {
469 const auto column =
470 std::find(header.begin(), header.end(), column_name);
471 svmp::throw_if<svmp::ParseException>(
472 column == header.end(),
473 "IonicModel reference CSV is missing state column " + column_name +
474 ": " + reference_path_);
475 state_columns.push_back(
476 static_cast<std::size_t>(std::distance(header.begin(), column)));
477 }
478 }

480 /// Return the pointwise tolerance @f$\mathrm{tol}(1+|expected|)@f$.
481 double scaled_tolerance(double expected) const
482 {
483 return configuration_.tolerance * (1.0 + std::fabs(expected));
484 }
485
486 /// Compare the checkpoint matching @p completed_updates, if one is listed.
487 void compare_checkpoint_if_present(std::size_t completed_updates,
488 const Vector<double> &X,
489 const Vector<double> &Xg,
490 std::size_t &checkpoint_index) const
491 {
492 if (checkpoint_index >= checkpoints_.size() ||
493 checkpoints_[checkpoint_index].completed_updates != completed_updates)
494 return;
495
496 const Checkpoint &checkpoint = checkpoints_[checkpoint_index];
497 std::size_t expected_index = 0;
498 for (std::size_t state = 0; state < X.size(); ++state, ++expected_index) {
499 const double expected = checkpoint.state[expected_index];
500 EXPECT_NEAR(X[state], expected, scaled_tolerance(expected))
501 << "checkpoint after " << checkpoint.completed_updates
502 << " completed updates, X[" << state << "] column "
503 << configuration_.state_reference_columns[expected_index];
504 }
505 for (std::size_t state = 0; state < Xg.size(); ++state, ++expected_index) {
506 const double expected = checkpoint.state[expected_index];
507 EXPECT_NEAR(Xg[state], expected, scaled_tolerance(expected))
508 << "checkpoint after " << checkpoint.completed_updates
509 << " completed updates, Xg[" << state << "] column "
510 << configuration_.state_reference_columns[expected_index];
511 }
512 ++checkpoint_index;
513 }

515 /// Configured concrete model stored behind the public IonicModel interface.
516 std::unique_ptr<IonicModel> model_;
517
518 /// Validated trajectory inputs and reference-column mapping.
519 IonicModelTrajectoryConfiguration configuration_;
520
521 /// Reference CSV path resolved beneath UNIT_TEST_DATA_DIR.
522 std::string reference_path_;
523
524 /// Strictly increasing reference checkpoints loaded from the CSV.
525 std::vector<Checkpoint> checkpoints_;
526};
