52 /// @brief Read-only Eigen view of an Array, sharing its storage.
53 ///
54 /// @tparam rows Row count, fixed at compile time; the columns are taken from the Array.
55 template <int rows>
56 Eigen::Map<const Eigen::Matrix<double, rows, Eigen::Dynamic>>
57 eigen_view(const Array<double>& A) {
58 if (A.nrows() != rows) {
59 svmp::raise<svmp::FE::InvalidArgumentException>(
60 "A view of " + std::to_string(rows) + " rows was requested for an array with " +
61 std::to_string(A.nrows()) + " rows.");
62 }
63 return {A.data(), rows, A.ncols()};
64 }

72 /// @brief Read-only Eigen view of a band of rows of an Array, sharing its storage.
73 ///
74 /// Used where an Array contains several fields and only a contiguous band is viewed.
75 /// For example,
76 ///
77 /// A = [ A00, A01, A02 ; eigen_view_rows<2>(A, 1) = [ A10, A11, A12 ;
78 /// A10, A11, A12 ; A20, A21, A22 ]
79 /// A20, A21, A22 ;
80 /// A30, A31, A32 ]
81 ///
82 /// @param A The Array to view.
83 /// @param first Index of the band's first row within the Array.
84 /// @tparam rows Number of rows, fixed at compile time, typically nsd.
85 template <int rows>
86 Eigen::Map<const Eigen::Matrix<double, rows, Eigen::Dynamic>, 0, Eigen::OuterStride<>>
87 eigen_view_rows(const Array<double>& A, const int first) {
88 if (first < 0 || first + rows > A.nrows()) {
89 svmp::raise<svmp::FE::InvalidArgumentException>(
90 "A view of " + std::to_string(rows) + " rows starting at row " + std::to_string(first) +
91 " was requested for an array with " + std::to_string(A.nrows()) + " rows.");
92 }
93 return {A.data() + first, rows, A.ncols(), Eigen::OuterStride<>(A.nrows())};
94 }

141 template <typename MatrixType>
142 void copy_Dm(const MatrixType& mat, Array<double>& dest) {
143 if ((mat.rows() != dest.nrows()) || (mat.cols() != dest.ncols())) {
144 const std::string mat_dims = "(" + std::to_string(mat.rows()) + "x" + std::to_string(mat.cols()) + ")";

147 "The 'mat" + mat_dims + "' and 'dest" + dest_dims +

152 for (int j = 0; j < mat.cols(); ++j) {
153 dest(i, j) = mat(i, j);
154 }
155 }

219 /// @brief Matrix product with the operand shape supplied at compile time.
220 ///
221 /// Overloads of mat_mul rather than differently named helpers, so a call
222 /// site states the shape and otherwise reads exactly as before:
223 ///
224 /// @code
225 /// mat_mul(Dm, Bm.rslice(b), DBm); // runtime shape check
226 /// mat_mul<6, 6, 3>(Dm, Bm.rslice(b), DBm); // no check, same arguments
227 /// @endcode
228 ///
229 /// The generic mat_mul overload above will dispatch to this overload
230 /// when the shapes are known at compile time.
231 ///
232 /// @tparam M rows of A and of the result
233 /// @tparam K columns of A and rows of B, the contracted dimension
234 /// @tparam N columns of B and of the result
235 template <int M, int K, int N>
236 void mat_mul(const Array<double>& A, const Array<double>& B,
237 Array<double>& C)
238 {
239 Eigen::Map<const Eigen::Matrix<double, M, K>> a(A.data());
240 Eigen::Map<const Eigen::Matrix<double, K, N>> b(B.data());
241 Eigen::Map<Eigen::Matrix<double, M, N>> c(C.data());
242
243 c.noalias() = a * b;
244 }
245
246 /// @brief As above, but with the column count known only at run time.
247 ///
248 /// For operands with one column per element node, where the width depends on
249 /// the element type. The row counts are still compile-time, which is where
250 /// most of the benefit comes from.
251 template <int M, int K>
252 void mat_mul(const Array<double>& A, const Array<double>& B,
253 Array<double>& C)
254 {
255 Eigen::Map<const Eigen::Matrix<double, M, K>> a(A.data());
256 Eigen::Map<const Eigen::Matrix<double, K, Eigen::Dynamic>> b(B.data(), K, B.ncols());
257 Eigen::Map<Eigen::Matrix<double, M, Eigen::Dynamic>> c(C.data(), M, C.ncols());
258
259 c.noalias() = a * b;
260 }

272 /**
273 * @brief Contracts two 4th order tensors A and B over two dimensions.
274 *
275 * For example, if dimsA = {0, 1} and dimsB = {2, 3} this is
276 * C_klmn = A_ijkl B_mnij (sum over i, j)
277 *
278 * @tparam nsd Number of spatial dimensions; each tensor is nsd^4.
279 * @param[in] A,B Fourth order tensors to contract.
280 * @param[in] dimsA,dimsB Indices of the contracted dimensions of A and B.
281 * @return The contracted tensor.
282 */
283 template <int nsd>
284 Tensor<nsd>
285 double_dot_product(const Tensor<nsd>& A, const std::array<int, 2>& dimsA,
286 const Tensor<nsd>& B, const std::array<int, 2>& dimsB) {
287
288 // Fast path for dimsA = dimsB = {2,3}: C_ijmn = A_ijkl * B_mnkl.
289 if (dimsA[0] == 2 && dimsA[1] == 3 && dimsB[0] == 2 && dimsB[1] == 3) {
290 constexpr int N = nsd * nsd;
291 Tensor<nsd> C;
292 Eigen::Map<const Eigen::Matrix<double, N, N>> a(A.data());
293 Eigen::Map<const Eigen::Matrix<double, N, N>> b(B.data());
294 Eigen::Map<Eigen::Matrix<double, N, N>> c(C.data());
295 c.noalias() = a * b.transpose();
296 return C;
297 }
298
299 // Define the contraction dimensions
300 Eigen::array<Eigen::IndexPair<int>, 2> contractionDims = {
301 Eigen::IndexPair<int>(dimsA[0], dimsB[0]), // Contract A's dimsA[0] with B's dimsB[0]
302 Eigen::IndexPair<int>(dimsA[1], dimsB[1]) // Contract A's dimsA[1] with B's dimsB[1]
303 };
304
305 // Return the double dot product
306 return A.contract(B, contractionDims);
307 }

311 /**
312 * @brief Compute the dyadic product of two 2nd order tensors A and B, C_ijkl = A_ij * B_kl
313 *
314 * @tparam nsd, the number of spatial dimensions
315 * @param A, the first 2nd order tensor
316 * @param B, the second 2nd order tensor
317 * @return Tensor<nsd>
318 */
319 template <int nsd>
320 Tensor<nsd>
321 dyadic_product(const Matrix<nsd>& A, const Matrix<nsd>& B) {
322 // Initialize the result tensor
323 Tensor<nsd> C;
324 constexpr int N = nsd * nsd;
325
326 // Column-major storage flattens index pairs: c(ij,kl) = a(ij) * b(kl).
327 Eigen::Map<const Eigen::Matrix<double, N, 1>> a(A.data());
328 Eigen::Map<const Eigen::Matrix<double, N, 1>> b(B.data());
329 Eigen::Map<Eigen::Matrix<double, N, N>> c(C.data());
330 c.noalias() = a * b.transpose();
331 return C;
332 }

336 /**
337 * @brief Create a 4th order identity tensor:
338 * I_ijkl = 0.5 * (δ_ik * δ_jl + δ_il * δ_jk)
339 *
340 * @tparam nsd, the number of spatial dimensions
341 * @return Tensor<nsd>
342 */
343 template <int nsd>
344 Tensor<nsd>
345 fourth_order_identity() {
346 // Initialize as zero
347 Tensor<nsd> I;
348 I.setZero();
349
350 // Set only non-zero entries
351 for (int i = 0; i < nsd; ++i) {
352 for (int j = 0; j < nsd; ++j) {
353 I(i,j,i,j) += 0.5;
354 I(i,j,j,i) += 0.5;
355 }
356 }
357
358 return I;
359 }

365 /// @brief Create a 4th order tensor from symmetric outer product of two matrices: C_ijkl = 0.5 * (A_ik * B_jl + A_il * B_jk)
366 ///
367 /// @tparam nsd Number of spatial dimensions.
368 /// @param[in] A,B Second order tensors.
369 /// @return The resulting 4th order tensor.
370 template <int nsd>
371 Tensor<nsd>
372 symmetric_dyadic_product(const Matrix<nsd>& A, const Matrix<nsd>& B) {
373
374 // Initialize the result tensor
375 Tensor<nsd> C;
376
377 // Compute the symmetric product: C_ijkl = 0.5 * (A_ik * B_jl + A_il * B_jk)
378 for (int l = 0; l < nsd; ++l) {
379 for (int k = 0; k < nsd; ++k) {
380 // blk views the (k,l) block of C, so blk(i,j) is C(i,j,k,l).
381 Eigen::Map<Eigen::Matrix<double, nsd, nsd>> blk(C.data() + nsd * nsd * (k + nsd * l));
382 blk.noalias() = 0.5 * (A.col(k) * B.col(l).transpose()
383 + A.col(l) * B.col(k).transpose());
384 }
385 }
386
387 // Return the symmetric product
388 return C;
389 }

393 /**
394 * @brief Performs a tensor transpose operation on a 4th order tensor A, B_ijkl = A_klij
395 *
396 * @tparam nsd, the number of spatial dimensions
397 * @param A, the input 4th order tensor
398 * @return Tensor<nsd>
399 */
400 template <int nsd>
401 Tensor<nsd>
402 transpose(const Tensor<nsd>& A) {
403
404 // Initialize the result tensor
405 Tensor<nsd> B;
406
407 // Permute the tensor indices to perform the transpose operation
408 for (int i = 0; i < nsd; ++i) {
409 for (int j = 0; j < nsd; ++j) {
410 for (int k = 0; k < nsd; ++k) {
411 for (int l = 0; l < nsd; ++l) {
412 B(i,j,k,l) = A(k,l,i,j);
413 }
414 }
415 }
416 }
417
418 return B;
419 }
