11/**
12 * @brief Uniform and time dependent active stress model.
13 *
14 * Defines an active tension that is constant in space but time dependent, i.e.
15 * @f[
16 * \Tact(t, \calcium, \fiberstretch, \fiberstretchrate, \astressstate) =
17 * g(t)\;,
18 * @f]
19 * where @f$g(t)@f$ is a user-defined function of time.
20 */
21class ActiveStressUniformUnsteady : public ActiveStress {
22public:
23 /// Model label.

25
26 /// Model parameters class.
27 class Parameters : public ActiveStressModelParameters {
28 public:
29 Parameters() : ActiveStressModelParameters(label) {
30 constexpr bool required = true;
31
32 add_parameter("Ramp", false, required);
33 add_parameter("Temporal_values_file_path", std::string(""), required);
34 }
35 };
