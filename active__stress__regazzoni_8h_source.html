11/**
12 * @brief Mean-field active stress model (implements the RDQ20-MF formulation).
13 *
14 * This class implements the mean-field RDQ20-MF sarcomere model of cardiomyocyte
15 * force generation of Regazzoni, Dede', and Quarteroni (2020), described in [1]
16 * and validated against the authors' reference implementation [2]. The node-local state has 20 variables: 16
17 * regulatory-unit (RU) probabilities (entries 0-15) describing the
18 * tropomyosin/troponin configuration of a triplet of neighbouring units, and 4
19 * crossbridge (XB) moments (entries 16-19). The RU probabilities are advanced
20 * with an explicit forward-Euler substepping scheme — for every macro time step,
21 * a number of smaller sub-steps are taken to update the RU states — and the XB
22 * moments with one implicit-Euler step per time step; the active tension is then
23 * reconstructed from the XB first moments.
24 *
25 * The returned scalar active tension is
26 * @f[
27 * \Tact = a_\text{XB} \, (\mu_P^1 + \mu_N^1) \, \phi(SL)\;,
28 * @f]
29 * where @f$\mu_P^1@f$ and @f$\mu_N^1@f$ are the permissive and non-permissive
30 * first XB moments (state entries 17 and 19), @f$\phi(SL)@f$ is the single-overlap
31 * fraction of the sarcomere at sarcomere length @f$SL = SL_0 \, \fiberstretch@f$
32 * (with @f$\fiberstretch@f$ the fiber stretch), and @f$a_\text{XB}@f$ is the tension
33 * upscaling factor. Because @f$\mu_P^1 + \mu_N^1@f$ and @f$\phi(SL)@f$ are
34 * dimensionless, @f$a_\text{XB}@f$ sets the units of the returned active tension.
35 *
36 * **References**:
37 * 1. [Regazzoni, Dede', Quarteroni (2020)](https://doi.org/10.1371/journal.pcbi.1008294)
38 * 2. [F. Regazzoni, cardiac-activation reference implementation](https://github.com/FrancescoRegazzoni/cardiac-activation)
39 *
40 * @note Although this model is governed by a system of ODEs, it inherits from
41 * @c ActiveStress rather than @c ActiveStressODE because it requires a customized
42 * time-stepping scheme to handle the stiffness of the model.
43 *
44 * @todo Force-strain-rate feedback requires a stabilization strategy for robust
45 * use in coupled electromechanics. This will be addressed in a follow-up PR.
46 */
47class RegazzoniActiveStress : public ActiveStress {
48public:
49 /// Model label, used for factory registration and XML selection.
50 static inline const std::string label = "Regazzoni";
51
52 /// @name State vector layout
53 /// @{
54
55 /// Number of regulatory-unit (RU) probability states (entries 0-15).
56 static constexpr unsigned int n_ru_states = 16;
57
58 /// Number of crossbridge (XB) moment states (entries 16-19).
59 static constexpr unsigned int n_xb_states = 4;
60
61 /// Total number of state variables.
62 static constexpr unsigned int n_state_variables = n_ru_states + n_xb_states;
63
64 /**
65 * @brief Flat index of the RU probability state P(TL, TC, TR, CC).
66 *
67 * Each argument is 0 or 1 and denotes the state of, respectively, the left
68 * tropomyosin unit, the central tropomyosin unit, the right tropomyosin unit
69 * and the central troponin (calcium unbound/bound). The ordering matches the
70 * reference implementation's serialization (TL outermost, CC innermost) and
71 * spans [0, 15].
72 */
73 static constexpr unsigned int ru_index(unsigned int TL, unsigned int TC,
74 unsigned int TR, unsigned int CC) {
75 return 8 * TL + 4 * TC + 2 * TR + CC;
76 }
77
78 /// Flat index of the XB moment state @p i (in [0, 3]), spanning [16, 19].
79 static constexpr unsigned int xb_index(unsigned int i) {
80 return n_ru_states + i;
81 }
82
83 /// @}
84

186private:
187 /// Array indexed over the four binary RU configuration variables (TL, TC, TR, CC).
188 using RUArray =
189 std::array<std::array<std::array<std::array<double, 2>, 2>, 2>, 2>;
190
191 /// Array indexed over a pair of binary state variables.
192 using BinaryPairArray = std::array<std::array<double, 2>, 2>;
193

195 using XBArray = std::array<double, 4>;

197 /// @name Regulatory-unit (RU) dynamics helpers
198 /// @{
199
200 /**
201 * @brief Compute the central-tropomyosin transition rate for each local RU
202 * configuration.
203 *
204 * Returns an @c RUArray where entry @c [TL][TC][TR][CC] is the rate at which
205 * the central tropomyosin changes state for that configuration. Because the
206 * rate depends on the neighbour states TL and TR, nearest-neighbour
207 * cooperativity is retained through the tracked TL-TC-TR configuration.
208 * These rates depend only on the model parameters, not on calcium or stretch.
209 *
210 * @return Central-tropomyosin transition rates, indexed @c [TL][TC][TR][CC].
211 */
212 RUArray ru_transition_rates_tropomyosin() const;
213
214 /**
215 * @brief Advance the 16 RU-state probabilities by one forward-Euler substep.
216 *
217 * Computes the probability fluxes caused by central-state transitions and the
218 * effective boundary-neighbour transitions from the mean-field closure, then
219 * updates @p state_RU in place.
220 *
221 * @param[in] dt Substep size [time].
222 * @param[in] rates_T Central-tropomyosin transition rates,
223 * indexed @c rates_T[TL][TC][TR][CC].
224 * @param[in] rates_C Troponin transition rates, indexed @c rates_C[CC][TC].
225 * @param[in,out] state_RU The 16 RU-state probabilities,
226 * indexed @c state_RU[TL][TC][TR][CC].
227 */
228 void ru_forward_euler_substep(double dt,
229 const RUArray &rates_T,
230 const BinaryPairArray &rates_C,
231 RUArray &state_RU) const;
232
233 /**
234 * @brief Advance the four crossbridge moments by one implicit-Euler step.
235 *
236 * Computes the permissivity and the effective permissive/non-permissive
237 * transition rates from the updated RU probabilities, forms the 4x4 linear
238 * system for the implicit update, and returns the updated moments.
239 *
240 * @param[in] dt Outer time step [time].
241 * @param[in] velocity Shortening velocity @f$-\dot{SL}/SL_0@f$ [1/time].
242 * @param[in] rates_T Central-tropomyosin transition rates,
243 * indexed @c rates_T[TL][TC][TR][CC].
244 * @param[in] state_RU The updated 16 RU-state probabilities,
245 * indexed @c state_RU[TL][TC][TR][CC].
246 * @param[in] state_XB The four crossbridge moments (input), ordered
247 * @f$[\mu_P^0, \mu_P^1, \mu_N^0, \mu_N^1]@f$.
248 * @return Updated crossbridge moments, ordered
249 * @f$[\mu_P^0, \mu_P^1, \mu_N^0, \mu_N^1]@f$.
250 */
251 XBArray xb_implicit_update(double dt, double velocity,
252 const RUArray &rates_T,
253 const RUArray &state_RU,
254 const XBArray &state_XB) const;
255
256 /**
257 * @brief Single-overlap fraction of the sarcomere at a given length.
258 *
259 * Returns the fraction @f$\phi(SL) \in [0, 1]@f$ of the sarcomere over which
260 * thin and thick filaments overlap exactly once, a piecewise-linear function
261 * of the sarcomere length built from the filament geometry (LA, LM, LB).
262 *
263 * @param[in] sarcomere_length Sarcomere length @f$SL@f$ [length].
264 */
265 double fraction_single_overlap(double sarcomere_length) const;

303 /// - @c false (default): use @f$v = -\dot{\lambda}@f$ as shortening velocity.
