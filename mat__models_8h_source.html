81/// @brief Viscous 2nd Piola-Kirchhoff stress and tangent contributions at a
82/// Gauss point, for the viscosity model configured for the domain.
83///
84/// Owns the buffers it computes into, so a kernel can hold one instance across
85/// an element's Gauss points and skip the calls it does not need.
86///
87/// @tparam nsd Number of spatial dimensions.
88///
89template <int nsd>
90class ViscousResponse {
91 public:
92 /// @brief Evaluate the domain's viscosity model at this Gauss point.
93 ///
94 /// Does nothing when recompute is false, preserving values from the
95 /// previous call. Used for linear triangules and tetrahedra for which
96 /// Nx, vx, and F are constant across the element's Gauss points.
97 ///
98 /// @param[in] lDmn Domain, supplying the viscosity model and its parameters.
99 /// @param[in] eNoN Number of element nodes.
100 /// @param[in] Nx Shape function spatial derivatives.
101 /// @param[in] vx Velocity gradient.
102 /// @param[in] F Deformation gradient.
103 /// @param[in] recompute Whether the stored values must be recomputed.
104 void update(const dmnType& lDmn, const int eNoN, const Array<double>& Nx,
105 const Matrix<nsd>& vx, const Matrix<nsd>& F, const bool recompute);
106
107 /// @brief Viscous 2nd Piola-Kirchhoff stress.
108 const Matrix<nsd>& S() const { return Svis_; }
109
110 /// @brief Tangent w.r.t. displacement. du(i*nsd + j, a, b) is the (i,j)
111 /// entry of the block coupling nodes a and b.
112 double du(const int ij, const int a, const int b) const { return Kvis_u_(ij, a, b); }
113
114 /// @brief Tangent w.r.t. velocity.
115 double dv(const int ij, const int a, const int b) const { return Kvis_v_(ij, a, b); }
116
117 private:
118 Matrix<nsd> Svis_;
119 Array3<double> Kvis_u_, Kvis_v_;
120};
121};
